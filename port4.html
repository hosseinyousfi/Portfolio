models.py 
from django.db import models

class Order(models.Model):
    customer = models.ForeignKey(
        User, on_delete=models.CASCADE
    )
    total = models.DecimalField(
        max_digits=10, decimal_places=0
    )

    def __str__(self):
        return f'سفارش #{self.id}'
